# Import required Spark SQL libraries
import org.apache.spark.sql._
import org.apache.spark.sql.functions._
import org.apache.spark.sql.types._
import org.apache.spark.sql.Row
import spark.implicits._

# 1. Create an Empty DataFrame
val schema = StructType(Array(
  StructField("ID", IntegerType, true),
  StructField("Name", StringType, true)
))

val emptyRDD = sc.emptyRDD[Row]

val emptyDF = spark.createDataFrame(emptyRDD, schema)

emptyDF.show()

# 2. Create an Empty Dataset
case class Student(id: Int, name: String)

val emptyDS = spark.emptyDataset[Student]

emptyDS.show()

# 3. Rename Nested Column
val data = Seq(
  (1, ("Rahul", 22)),
  (2, ("Priya", 21))
)

val df = data.toDF("ID", "Student")

val nestedDF = df.select(
  col("ID"),
  struct(
    col("Student._1").as("Name"),
    col("Student._2").as("Age")
  ).as("StudentInfo")
)

nestedDF.show(false)

# 4. Adding or Updating a Column on DataFrame
val empDF = Seq(
  (1, "Rahul", 50000),
  (2, "Priya", 60000),
  (3, "Amit", 55000)
).toDF("ID", "Name", "Salary")

empDF.show()

# Add Bonus column
val empDF1 = empDF.withColumn("Bonus", col("Salary") * 0.10)

empDF1.show()

# Increase Salary by 5000
val empDF2 = empDF.withColumn("Salary", col("Salary") + 5000)

empDF2.show()

# 5. Drop a Column on DataFrame
val dropDF = empDF.drop("Salary")

dropDF.show()

# 6. Adding a Literal Constant to DataFrame
val countryDF = empDF.withColumn("Country", lit("India"))

countryDF.show()

# 7. Changing Column Data Type
val salaryDF = Seq(
  ("1", "50000"),
  ("2", "60000"),
  ("3", "55000")
).toDF("ID", "Salary")

val salaryDF1 = salaryDF.withColumn(
  "Salary",
  col("Salary").cast(IntegerType)
)

salaryDF1.printSchema()

salaryDF1.show()

# 8. Pivot a DataFrame
val salesData = Seq(
  ("Sales", "Jan", 1000),
  ("Sales", "Feb", 1500),
  ("HR", "Jan", 800),
  ("HR", "Feb", 1200)
).toDF("Department", "Month", "Salary")

salesData.show()

val pivotDF = salesData
  .groupBy("Department")
  .pivot("Month")
  .sum("Salary")

pivotDF.show()

# 9. Unpivot a DataFrame
val unpivotData = Seq(
  ("Sales", 1000, 1500),
  ("HR", 800, 1200)
).toDF("Department", "Jan", "Feb")

unpivotData.show()

val unpivotDF = unpivotData.selectExpr(
  "Department",
  "stack(2, 'Jan', Jan, 'Feb', Feb) as (Month, Salary)"
)

unpivotDF.show()

# 10. Create a DataFrame using StructType and StructField Schema
val empSchema = StructType(Array(
  StructField("ID", IntegerType, true),
  StructField("Name", StringType, true),
  StructField("Age", IntegerType, true),
  StructField("Salary", DoubleType, true)
))

# Create Row Data
val empRows = Seq(
  Row(1, "Rahul", 22, 50000.0),
  Row(2, "Priya", 21, 60000.0),
  Row(3, "Amit", 23, 55000.0)
)

# Convert Row Data into RDD
val empRDD = sc.parallelize(empRows)

# Create DataFrame using Schema
val empSchemaDF = spark.createDataFrame(empRDD, empSchema)

# Display Schema
empSchemaDF.printSchema()

# Display DataFrame
empSchemaDF.show()

# Practical 2B

# Import Required Libraries
import org.apache.spark.sql._
import org.apache.spark.sql.functions._
import spark.implicits._

# 1. Selecting the First Row of Each Group
val data = Seq(
  (1,"Amit","IT",50000),
  (2,"Priya","IT",60000),
  (3,"Rahul","HR",45000),
  (4,"Neha","HR",55000),
  (5,"Ravi","Sales",40000)
)

val df = data.toDF("ID","Name","Department","Salary")

df.show()

import org.apache.spark.sql.expressions.Window

val windowSpec = Window.partitionBy("Department").orderBy("Salary")

val result = df.withColumn("RowNum", row_number().over(windowSpec))

result.filter($"RowNum" === 1).show()

# 2. Sort DataFrame
val df = Seq(
  (1,"Amit",50000),
  (2,"Priya",60000),
  (3,"Rahul",45000)
).toDF("ID","Name","Salary")

df.sort("Salary").show()

df.sort(desc("Salary")).show()

# 3. Union DataFrames
val df1 = Seq(
  (1,"Amit"),
  (2,"Priya")
).toDF("ID","Name")

val df2 = Seq(
  (3,"Rahul"),
  (4,"Neha")
).toDF("ID","Name")

val result = df1.union(df2)

result.show()

# 4. Drop Rows with Null Values
val df = Seq(
  (1,"Amit",50000),
  (2,null.asInstanceOf[String],60000),
  (3,"Rahul",null.asInstanceOf[Integer])
).toDF("ID","Name","Salary")

df.show()

df.na.drop().show()

# 5. Split Single Column into Multiple Columns
val df = Seq(
  ("Amit,Mumbai"),
  ("Priya,Pune"),
  ("Rahul,Nagpur")
).toDF("Details")

val result = df
  .withColumn("Name", split($"Details",",")(0))
  .withColumn("City", split($"Details",",")(1))

result.show()

# 6. Concatenate Multiple Columns
val df = Seq(
  ("Amit","Sharma"),
  ("Priya","Patil")
).toDF("FirstName","LastName")

val result = df.withColumn(
  "FullName",
  concat_ws(" ",$"FirstName",$"LastName")
)

result.show()

# 7. Replace Null Values
val df = Seq(
  (1,"Amit"),
  (2,null.asInstanceOf[String]),
  (3,"Rahul")
).toDF("ID","Name")

df.na.fill("Unknown").show()

# 8. Remove Duplicate Rows
val df = Seq(
  (1,"Amit"),
  (2,"Priya"),
  (2,"Priya"),
  (3,"Rahul")
).toDF("ID","Name")

df.dropDuplicates().show()

# 9. Remove Duplicates Based on Selected Columns
val df = Seq(
  (1,"Amit"),
  (1,"Amit Kumar"),
  (2,"Priya"),
  (2,"Priya")
).toDF("ID","Name")

df.dropDuplicates("ID").show()

df.dropDuplicates("ID","Name").show()

# 10. Spark UDF
import org.apache.spark.sql.functions.udf

val upperCase = udf((name:String) => name.toUpperCase)

val df = Seq(
  ("amit"),
  ("priya"),
  ("rahul")
).toDF("Name")

val result = df.withColumn(
  "UpperName",
  upperCase($"Name")
)

result.show()
